# 라즈베리파이(데몬 노드)로 보낼 것 — deploy_pi.sh 가 이 파일을 읽는다.
#
# 목록을 스크립트 안에 숨기지 않고 여기 따로 두는 이유:
# git diff 에 "무엇을 파이로 보내는가"의 변화가 그대로 보인다. 스크립트
# 안의 --exclude 플래그로만 두면 새 폴더가 생겼을 때 아무도 못 알아챈다.
#
# 규칙
#   include <경로>   보낸다 (디렉터리면 통째로)
#   exclude <패턴>   include 안에서 뺀다
#   빈 줄과 # 은 무시
#
# ============================================================
#  누가 무엇을 쓰나  (import 사슬을 따라가 확인, 2026-08-21)
# ============================================================
#
# 원칙: 하드웨어를 만지면 파이, 화면을 그리면 PC.
#
# 파이 = grasp_daemon.py 가 실제로 import 하는 23개
#   detection/     grasp_daemon grasp_commands grasp_state
#                  link link_sender link_watchdog
#                  orientation roi_config roi_judge wrist_align
#   hand_control/  hand hand_config kinematics sequence servo_bus wrist
#                  tactile grasp grasp_log grasp_runner
#                  force_control stiffness spread_seek
#
# 파이에 같이 두면 좋은 것 (데몬은 안 쓰지만 하드웨어가 파이에 있다)
#   hand_control/grasp_main.py      --sensor-diag / --sensor-only / 수동 파지
#   hand_control/main.py            a/s 키로 손 직접 제어
#   hand_control/calibrate_open.py  서보 영점 캘리브레이션
#   hand_control/servo_id_tool.py   서보 ID 설정
#   detection/roi_grasp.py          단일 프로세스 기준선 (카메라+모터)
#
# PC 전용 = grasp_console.py 사슬
#   detection/grasp_console.py  console_input.py  dashboard/
#
# ============================================================
#  보내는 것
# ============================================================
#
# 폴더째 보낸다. 파일을 하나씩 나열하면 새 모듈이 생겼을 때 빠뜨리고,
# 그 증상은 파이에서 ImportError 로만 나타난다 -- 위 목록에 이미
# spread_seek.py 처럼 나중에 생긴 파일이 들어 있다.
#
# PC 전용 파일(콘솔·대시보드)도 같이 가지만 50KB 남짓이고 파이에서는
# 아무도 import 하지 않는다. 빠뜨릴 위험보다 이게 싸다.
include detection
include hand_control

# 모터 ID 와 영점 오프셋. 이 저장소가 이 파일의 주인이다.
# 없으면 데몬이 load_fingers 에서 FileNotFoundError 로 죽는다.
include config

include requirements.txt
include docs

# ============================================================
#  안 보내는 것 (일부러)
# ============================================================
#
# vendor/  (Tashan capRead SDK)
#   파이에서 CH341 커널 모듈을 직접 빌드했고 class_ch341.py 를 aarch64
#   용으로 고쳐 뒀다(docs/pi-setup.md 3장). 덮으면 그 수정이 날아간다.
#   파이의 위치가 다르면 CAPREAD_DIR 로 알려준다.
#
# venv/
#   파이는 aarch64 휠을 쓴다. 자기 것을 그대로 둔다.

# ============================================================
#  include 안에서 빼는 것
# ============================================================

# 파이는 Python 3.13, PC 는 3.12 라 .pyc 를 못 쓴다. 게다가 옛 .pyc 가
# 남아 있으면 어느 소스가 도는지 헷갈린다.
exclude __pycache__
exclude *.pyc
exclude .pytest_cache

# PC 의 분석용 로그다. 파이는 자기 로그를 따로 쓴다.
# scp -r 이 이것까지 보내다 전송이 실패한 적이 있다(2026-08-21).
exclude logs

# 실행 중에 만들어지는 캘리브레이션. 설치마다 다르다 -- 파이가 자기
# ROI 와 손 마스크를 갖고 있는데 PC 것으로 덮으면 조용히 어긋난다.
exclude roi.json
exclude hand_mask.npy

# 모델 가중치. 각도는 깊이 밴드의 주축(PCA)으로 낸다(orientation.py).
exclude *.pt
